Все команды консоли:
  builders,bs                   Показать список форм

  make [builder]                Запустить форму создания миграции
    --field1=[value]            Задать значения поля
    ...
    --fieldN=[value]            Задать значения поля

  list,ls                       Показать список миграций
    --new                       Показать только новые
    --installed                 Показать только установленные
    --search=[text]             Отфильтровать по названию, описанию или тегу
    --tag=[text]                Отфильтровать по тегу

  up                            Установить все новые миграции
    --search=[text]             Отфильтровать по названию, описанию или тегу
    --add-tag=[text]            Задать тег

  up [version1, ...]            Установить выбранную миграцию
    --add-tag=[text]            Задать тег

  down                          Откатить все установленные миграции
    --search=[text]             Отфильтровать по названию, описанию или тегу
    --tag=[text]                Отфильтровать по тегу

  down [version1, ...]          Откатить выбранную миграцию

  redo [version1, ...]          Откатить и установить выбранную миграцию

  mark [version1, ...]          Отметить миграцию
    --new                       Отметить все новые миграции
    --installed                 Отметить все установленные миграции
    --unknown                   Отметить все неизвестные миграции
    --as=installed              Как установленную (не запуская ее)
    --as=new                    Как новую (не откатывая ее)
    --add-tag=[text]            Задать тег (если установленная)

  settag [version1, ...]        Задать тег миграции
    --installed                 Задать тег всем установленным миграциям
    --unknown                   Задать тег всем неизвестным миграциям
    --add-tag=[text]            Название тега

  move [version, ...]           Переместить миграции в другой конфиг
   --new                        Переместить все новые миграции
   --installed                  Переместить все установленные миграции
   --unknown                    Переместить все неизвестные миграции
   --to=[config]                Название конфига

  delete,del [version, ...]     Удалить миграцию
    --new                       Удалить все новые миграции
    --installed                 Удалить все установленные миграции
    --unknown                   Удалить все неизвестные миграции

  config                        Показать текущий конфиг
    --config=[name]             Переключить конфиг на выбранный

  help                          Помощь по командам
